// YOUR ACTUAL PUBLISHED WEBSITE

const BASE_URL =
"https://digitalartistic6494-eng.github.io/R-d-Qr-generator/";


// CREATE UNIQUE MESSAGE LINK

function createLink(title, message) {

  const data = {
    t: title,
    m: message
  };

  const encoded = btoa(
    unescape(
      encodeURIComponent(JSON.stringify(data))
    )
  );

  return BASE_URL + "#note=" +
    encodeURIComponent(encoded);

}


// GENERATE QR CODE

document.getElementById("form")
.addEventListener("submit", function(event) {

  event.preventDefault();

  if (typeof QRCode === "undefined") {
    alert("QR library not loaded. Check internet.");
    return;
  }

  const title =
    document.getElementById("title").value.trim()
    || "R&D Message";

  const message =
    document.getElementById("message").value.trim();

  if (!message) {
    alert("Please enter a message.");
    return;
  }

  if (message.length > 200) {
    alert("Maximum message length is 200 characters.");
    return;
  }

  const url = createLink(title, message);

  const card = document.createElement("div");
  card.className = "qr-card";

  const image = document.createElement("div");
  image.className = "qr-image";

  const details = document.createElement("div");
  details.className = "qr-details";

  const heading = document.createElement("h3");
  heading.textContent = title;

  const description = document.createElement("p");
  description.textContent = message;

  const actions = document.createElement("div");
  actions.className = "actions";


  // DOWNLOAD QR

  const download = document.createElement("button");
  download.textContent = "DOWNLOAD QR";

  download.addEventListener("click", function() {

    const canvas = image.querySelector("canvas");

    if (!canvas) {
      alert("QR code is not ready.");
      return;
    }

    const link = document.createElement("a");

    link.download = "Bactolac-RD-QR.png";

    link.href = canvas.toDataURL("image/png");

    link.click();

  });


  // COPY MESSAGE LINK

  const copy = document.createElement("button");

  copy.textContent = "COPY LINK";
  copy.className = "secondary";

  copy.addEventListener("click", async function() {

    try {

      await navigator.clipboard.writeText(url);

      copy.textContent = "COPIED!";

    } catch {

      prompt("Copy this message URL:", url);

    }

  });


  // CREATE QR IMAGE

  new QRCode(image, {

    text: url,

    width: 280,
    height: 280,

    colorDark: "#000000",
    colorLight: "#ffffff",

    correctLevel: QRCode.CorrectLevel.H

  });


  actions.append(download, copy);

  details.append(heading, description, actions);

  card.append(image, details);

  document.getElementById("results").prepend(card);

  document.getElementById("empty").style.display = "none";

  document.getElementById("form").reset();

});

</script>
